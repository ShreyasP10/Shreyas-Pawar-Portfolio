// app/page.tsx
"use client";
import { useState } from "react";
import Sidebar from "@/components/Sidebar";
import Navigation from "@/components/Navigation";
import About from "@/components/About";
import Projects from "@/components/Projects";
import Experience from "@/components/Experience";
import Contact from "@/components/Contact";

export default function Home() {
  const [activeTab, setActiveTab] = useState("About Me");

  // Render content based on the active tab
  const renderContent = () => {
    switch(activeTab) {
      case "About Me": return <About />;
      case "Project": return <Projects />;
      case "Experience": return <Experience />;
      case "Contact me": return <Contact />;
      default: return <About />;
    }
  };

  return (
    <main className="flex flex-col lg:flex-row min-h-screen bg-[#0F0F0F] text-[#FCA311] p-4 gap-4">
      {/* Left Sidebar - Static */}
      <aside className="lg:w-1/3 w-full h-full border-2 border-[#FCA311] rounded-2xl p-6 flex flex-col gap-6">
        <Sidebar />
      </aside>

      {/* Right Content Frame - Dynamic & Wrapping */}
      <section className="flex-1 w-full h-auto border-2 border-[#FCA311] rounded-2xl p-6 flex flex-col gap-4">
        <div className="flex flex-col md:flex-row justify-between items-start md:items-center border-b-2 border-[#FCA311] pb-4 mb-4">
           {/* Page Name dynamically changing */}
           <h2 className="text-3xl font-bold">{activeTab}</h2> 
           
           {/* Navigation */}
           <Navigation activeTab={activeTab} setActiveTab={setActiveTab} />
        </div>
        
        {/* Content Wraps/Scales based on children size */}
        <div className="flex-grow h-auto">
          {renderContent()}
        </div>
      </section>
    </main>
  );
}

A portfolio website isn't just a digital resume; its ultimate goal is conversion—convincing a recruiter or client that you are the right person to hire.

To achieve that, a portfolio needs a blend of content, functionality, design, and technical reliability. Here is the definitive breakdown of what a portfolio website truly needs:

1. Content & Information Architecture (The "What")
These are the non-negotiable sections that answer the fundamental questions a visitor has.

A Strong Bio / "Hero" Section: A clear, punchy one-liner stating who you are and what you do (e.g., "Full Stack Developer specializing in React and Next.js") alongside your photo.

Your Projects (The Core): This is the proof of your work. Each project needs a title, a short description, the technologies used (tech stack tags), and a link to a live demo or GitHub repository. Screenshots are vital—people trust visual evidence over text.

Skills & Tech Stack: A visual list of your tools (languages, frameworks, databases). This helps recruiters quickly scan if you match their job description.

Professional Experience: A condensed, chronological list of your past job roles, companies, and key achievements.

Contact Information & Call-to-Action (CTA): The most important part. A contact form, your email, and links to your social profiles (LinkedIn, GitHub). Crucially, it needs a clear CTA button, like "Hire Me", "Download My CV", or "Let's Connect".

2. Functional Requirements (The "What it does")
Smooth, Intuitive Navigation: As shown in your wireframe, visitors must instantly know how to move between sections. The navigation should be clearly visible and change state (highlighting the active tab) to tell the user where they are.

Functional Contact Form: The form must actually work. It needs client-side validation (making sure the email format is correct) and, most importantly, server-side handling to securely send the message to your inbox without exposing your email credentials to web scrapers.

Clickable Social & External Links: All your social icons must be anchor tags (<a href="...">) that open in a new tab (target="_blank"), allowing recruiters to easily cross-reference your profiles.

3. Design, UX, & Accessibility (The "How it feels")
Mobile Responsiveness: Over 50% of recruiters browse on their phones. Your content frame must "wrap" and stack vertically for mobile screens, exactly as you mentioned in your sketch.

Performance & Speed: A portfolio that takes 5 seconds to load will lose 50% of its visitors. Images must be compressed and lazy-loaded.

Visual Hierarchy & Branding: The colors, fonts, and layout must reflect your personality (e.g., your dark-mode wireframe with orange outlines is a fantastic, instantly recognizable "brand").

Accessibility (a11y): Good color contrast, keyboard-navigable menus, and alt text on images. A recruiter with a screen reader should still be able to navigate your site.

4. Technical & Infrastructure Needs (The "Under the hood")
A Custom Domain: yourname.com looks infinitely more professional than yourname.github.io or yourname.wixsite.com.

SSL/HTTPS: Security encryption is mandatory (most modern hosters provide this for free).

Analytics: You need to know which projects are attracting attention and where your traffic is coming from. Google Analytics 4 or privacy-friendly tools like Plausible or Umami are essential.

SEO (Search Engine Optimization): Proper meta titles, descriptions, and semantic HTML (<h1>, <h2>, etc.) so that when someone Googles your name, your portfolio shows up at the top.

5. The "Nice-to-Haves" (The extra edge)
A Blog: Writing articles about your tech stack (e.g., "How I built my portfolio with Next.js") proves your expertise and dramatically boosts your SEO.

Resume Download: A button that automatically downloads your PDF resume.

Dark/Light Mode: While your provided wireframe is dark-only, offering a toggle shows advanced frontend proficiency.

Subtle Animations: Micro-interactions (like hover effects on projects or a subtle fade-in when tabs change) make the site feel polished and premium.
